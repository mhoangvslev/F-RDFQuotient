summary table name: "summary_"<timestamp><info>"_"<summaryType>
rep table name: "summary_"<timestamp><info>"_"<summaryType>"_rep"
edges table name: "summary_"<timestamp><info>"_"<summaryType>"_edges"
node stats table name: "summary_"<timestamp><info>"_"<summaryType>"_nodeStats"

<timestamp>: datetime in yyyyMMdd_HHmmss format
<info>: "", "_sat"
<summaryType>: "t", "2pioat", "2p1fw", "2p1fb", "2ps", "2pw", "2pwuf", "2pts", "2ptw", "s", "w", "ts", "tw"
